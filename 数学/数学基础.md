## 最大公约数gcd,最小公倍数lcm
### 辗转相除法
```cpp
#include<iostream>
using namespace std;
int gcd(int da,int xiao){
    int temp;
    while(xiao!=0){
        temp=da%xiao;
        da=xiao;
        xiao=temp;
    }
    return da;
}
int main(){
    int a,b;
    cin>>a>>b;
    int lcm=a/gcd(a,b)*b;
    cout<<lcm;


    return 0;
}
```
## 快速幂
### 求A^B最后三位
### 递归
```cpp
int power(int a,int n){
    int ans=0;
    if(n==0)ans=1;
    else{
        ans=power(a*a,n/2);
        if(n%2==1)
        ans*=a;
    }
    return ans;
}
```
### 非递归
```cpp
int power(int a,int n){
    int ans=1;
    while(n){
        if(n%2)ans*=a;
        a=a*a;
        n=n/2;
    }
    return ans;
}
```
### 二分查找
```cpp
int BiSearch(int a[],int n,int x){
    int left=0,right=n-1;
    int middle=(left+right)/2;
    while(left<=right){
        if(middle==x){
            return middle;
        }
        else if(x>middle)
        return left=middle+1;
        else
        return left=middle-1;
    }
    return -1;
}
```

int Bisearch(int a[],int x,int left,int right){
    if(left>right)return -1;
    int middle=(left+right)/2;
    else{
        if (a[middle]==x)
        return middle;
        else if(x>middle)
        return (a,x,left+1,right);
